delete from pla_base_seguimiento_agencias_plazo_1
where periodo='202608';

insert into pla_base_seguimiento_agencias_plazo_1

select
'202608' periodo,
(case when a.codigoclientebanco is not null then a.codigoclientebanco else b.codcli_cancelado end) codigoclientebanco,
(case when a.codigoclientebanco is not null then 1 else 0 end) c_vence,
(case when b.codcli_cancelado is not null then 1 else 0 end) c_cancelado,
(case when b.codcli_aperturado is not null then 1 else 0 end) c_aperturado,
(case when a.codigoclientebanco is not null then saldo_x_vencer else 0 end) s_vence,
(case when b.codcli_cancelado is not null then saldo_cancelado else 0 end) s_cancelado,
(case when b.codcli_aperturado is not null then saldo_aperturado else 0 end) s_aperturado

from
(
    select
        codigoclientebanco,
        sum(saldoactual * tipocambio) saldo_x_vencer
    from maesaldos_260731
    where producto like 'plazo%'
      and SubProducto <> 'CTS'
      and flagstk > 0
      and format(fechavencimientooriginal,'yyyyMM') = '202608'
    group by codigoclientebanco
) a

full join
(
    select
        aa.codigoclientebanco codcli_cancelado,
        saldo_cancelado,
        bb.codigoclientebanco codcli_aperturado,
        saldo_aperturado
    from
    (
        select
            codigoclientebanco,
            sum(saldocancelacion * tipocambio) saldo_cancelado
        from maesaldos_260831
        where producto like 'plazo%'
          and SubProducto <> 'CTS'
          and format(fechacancelacion,'yyyyMM') = '202608'
          and format(fechacancelacion,'yyyyMM') <> format(fechaaperturaoriginal,'yyyyMM')
        group by codigoclientebanco
    ) aa

    left join

    (
        select
            codigoclientebanco,
            sum(saldoactual * tipocambio) saldo_aperturado
        from maesaldos_260831
        where producto like 'plazo%'
          and SubProducto <> 'CTS'
          and flagstk > 0
          and format(fechaaperturaoriginal,'yyyyMM') = '202608'
        group by codigoclientebanco
    ) bb
    on aa.codigoclientebanco = bb.codigoclientebanco
) b
on a.codigoclientebanco = b.codcli_cancelado;

DROP TABLE IF EXISTS pla_base_pasivos_saldo_v2;

SELECT
    periodo,
    moneda,
    producto,
    agencia_final,
    grupo_periodo_venta,
    zona,
    situacion,
    fecha_cierre,
    codigoclientebanco,
    NumeroOperacion,
    SaldoActual,
    saldo_solarizado,
    saldo_solarizado_tc_prox_mes,
    saldo_solarizado_tc_actual,
    Tasa_x_Saldo_Actual,
    Tasa_x_Saldo_Solarizado
INTO dbo.pla_base_pasivos_saldo_v2
FROM
(
    SELECT *,
           ROW_NUMBER() OVER
           (
                PARTITION BY periodo,numerooperacion
                ORDER BY periodo,numerooperacion
           ) rn
    FROM
    (
        SELECT
            x.*,
            zona,
            fecha_cierre,
            (
                CASE
                    WHEN fecha_cierre IS NULL THEN 'SE MANTIENE'
                    WHEN fecha_cierre>'2025-12-31' THEN 'VA A CERRAR'
                    WHEN fecha_cierre>='2024-12-31'
                      OR fecha_cierre<='2025-12-31'
                      OR categoria IS NULL THEN 'CERRADA'
                    WHEN fecha_cierre<'2024-12-31' THEN 'CERRADA ANTES'
                    ELSE 'OTROS'
                END
            ) situacion

        FROM #temp2 x

        LEFT JOIN
        (
            SELECT
                agencia,
                categoria,
                zona,
                fecha_cierre
            FROM pla_agencias_ic
            WHERE periodo='202608'
              AND canal_agencia='AGENCIAS'
              AND zona IS NOT NULL
        ) y

        ON (
            CASE
                WHEN x.agencia_final='AG. CARSA PUCALLPA'
                    THEN 'AG. PUCALLPA'
                WHEN x.agencia_final='AG. CARSA JAEN'
                    THEN 'AG. JAEN'
                ELSE x.agencia_final
            END
        )=y.agencia
    ) w
) x
WHERE rn=1;

                when y.fecha_cierre >= '2024-12-31'
                     or y.fecha_cierre <= '2025-12-31'
                     or y.categoria is null
                    then 'CERRADA'
                when y.fecha_cierre < '2024-12-31'
                    then 'CERRADA ANTES'
                else 'OTROS'
            end
        ) situacion,

        z.cartera

    from #temp2_a x

    left join
    (
        select
            agencia,
            categoria,
            zona,
            fecha_cierre
        from pla_agencias_ic
        where periodo='202608'
          and canal_agencia='AGENCIAS'
          and zona is not null
    ) y

    on (
        case
            when x.agencia_final='AG. CARSA PUCALLPA'
                then 'AG. PUCALLPA'
            when x.agencia_final='AG. CARSA JAEN'
                then 'AG. JAEN'
            else x.agencia_final
        end
    ) = y.agencia

    left join #temp_pld_cartera z
        on x.numerooperacion=z.numerooperacion
